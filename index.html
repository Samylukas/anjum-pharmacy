function doGet(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. Products Data var sheetProducts = ss.getSheetByName("جدول_المنتجات") || ss.getSheetByName("Products") || ss.getSheets()[0]; var productsData = sheetProducts ? sheetProducts.getDataRange().getValues() : []; var cloudProducts = []; var productNamesSet = {}; for (var k = 1; k < productsData.length; k++) { var pName = productsData[k][1] ? productsData[k][1].toString().trim() : ""; if (pName && !productNamesSet[pName]) { productNamesSet[pName] = true; cloudProducts.push({ n: pName, p: parseFloat(productsData[k][2]) || 0, q: parseInt(productsData[k][3]) || 0 }); } } // 2. Sales Data var sheetSales = ss.getSheetByName("جدول_البيع") || ss.getSheetByName("Sales"); var salesData = sheetSales ? sheetSales.getDataRange().getValues() : []; var resultInvoices = []; var invoiceIdsSet = {}; var maxId = 1000; for (var i = 1; i < salesData.length; i++) { var invId = parseInt(salesData[i][1]) || 0; if (invId > 0) { if (invId > maxId) maxId = invId; if (!invoiceIdsSet[invId]) { invoiceIdsSet[invId] = true; var formattedDate = salesData[i][0] || salesData[i][2]; if (formattedDate instanceof Date) { formattedDate = Utilities.formatDate(formattedDate, ss.getSpreadsheetTimeZone(), "yyyy-MM-dd HH:mm"); } resultInvoices.push({ id: invId, date: formattedDate ? formattedDate.toString() : "", itemsText: salesData[i][3] || "", usd: parseFloat(salesData[i][4]) || 0, eur: parseFloat(salesData[i][5]) || 0, gbp: parseFloat(salesData[i][6]) || 0, egp: parseFloat(salesData[i][7]) || 0, visa: parseFloat(salesData[i][8]) || 0, totalComm: parseFloat(salesData[i][9]) || 0, totalEGP: parseFloat(salesData[i][10]) || 0, netProfit: parseFloat(salesData[i][11]) || 0, status: 'Active' }); } } } // 3. Finance (Expenses & Revenues) Data var sheetFinance = ss.getSheetByName("جدول_المصروفات_والإيرادات"); var financeData = sheetFinance ? sheetFinance.getDataRange().getValues() : []; var resultFinance = []; for (var f = 1; f < financeData.length; f++) { var fId = parseInt(financeData[f][0]) || f; var fDate = financeData[f][1]; if (fDate instanceof Date) { fDate = Utilities.formatDate(fDate, ss.getSpreadsheetTimeZone(), "yyyy-MM-dd HH:mm"); } resultFinance.push({ id: fId, date: fDate ? fDate.toString() : "", type: financeData[f][2] || "Expense", category: financeData[f][3] || "", amount: parseFloat(financeData[f][4]) || 0, notes: financeData[f][5] || "" }); } var output = { status: "success", nextInvoiceId: maxId + 1, products: cloudProducts, invoices: resultInvoices, finances: resultFinance }; var callback = e.parameter ? e.parameter.callback : null; if (callback) { return ContentService.createTextOutput(callback + "(" + JSON.stringify(output) + ")") .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(JSON.stringify(output)).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.message })) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { var lock = LockService.getScriptLock(); try { lock.waitLock(10000); // Concurrency Prevention var ss = SpreadsheetApp.getActiveSpreadsheet(); var req = JSON.parse(e.postData.contents); var action = req.action; var d = req.data; function cleanName(str) { if (!str) return ""; return str.toString().trim().replace(/\s+/g, " "); } function cleanDateStr(dStr) { if (!dStr) return ""; return dStr.toString().replace('T', ' ').replace('.000Z', ''); } if (action === "updateSingleProduct") { var sheetProducts = ss.getSheetByName("جدول_المنتجات") || ss.insertSheet("جدول_المنتجات"); var lastRow = sheetProducts.getLastRow(); if (lastRow === 0) { sheetProducts.appendRow(["م", "اسم المنتج / الصنف", "السعر التجاري (EGP)", "الكمية المتاحة بالمخزن"]); lastRow = 1; } var data = sheetProducts.getRange(1, 1, lastRow, 4).getValues(); var targetName = cleanName(d.n); var found = false; for (var i = 1; i < data.length; i++) { if (cleanName(data[i][1]) === targetName && targetName !== "") { sheetProducts.getRange(i + 1, 3).setValue(parseFloat(d.p) || 0); sheetProducts.getRange(i + 1, 4).setValue(parseInt(d.q) || 0); found = true; break; } } if (!found && targetName !== "") { sheetProducts.appendRow([sheetProducts.getLastRow(), d.n, parseFloat(d.p) || 0, parseInt(d.q) || 0]); } } else if (action === "deleteProduct") { var sheetProducts = ss.getSheetByName("جدول_المنتجات") || ss.getSheetByName("Products"); if (sheetProducts) { var pData = sheetProducts.getDataRange().getValues(); var targetName = cleanName(d.name); for (var p = pData.length - 1; p >= 1; p--) { if (cleanName(pData[p][1]) === targetName) { sheetProducts.deleteRow(p + 1); } } } } else if (action === "saveInvoice") { var sheetSales = ss.getSheetByName("جدول_البيع") || ss.insertSheet("جدول_البيع"); if (sheetSales.getLastRow() === 0) { sheetSales.appendRow(["تاريخ الفاتورة الفعلي", "رقم الفاتورة", "تاريخ الإدخال", "تفاصيل الأصناف المباعة", "دولار ($)", "يورو (€)", "إسترليني (£)", "مصري (EGP)", "فيزا (Visa)", "إجمالي التجاري", "إجمالي البيع (EGP)", "صافي الربح", "الحالة"]); } var salesData = sheetSales.getDataRange().getValues(); var targetRow = -1; for (var k = 1; k < salesData.length; k++) { if (parseInt(salesData[k][1]) === parseInt(d.id)) { targetRow = k + 1; break; } } var itemsCleanText = ""; var u = 0, eu = 0, g = 0, eg = 0, v = 0; if (d.items && Array.isArray(d.items)) { itemsCleanText = d.items.map(function(i) { u += parseFloat(i.usd || 0); eu += parseFloat(i.eur || 0); g += parseFloat(i.gbp || 0); eg += parseFloat(i.egp || 0); v += parseFloat(i.visa || 0); return i.name + " (العدد: " + (i.qty || 1) + " - بيع: " + (parseFloat(i.sellTotal) || 0).toFixed(2) + " ج.م)"; }).join(" | "); } else if (typeof d.items === 'string') { itemsCleanText = d.items; } else if (d.itemsText) { itemsCleanText = d.itemsText; } var rowValues = [ cleanDateStr(d.date), parseInt(d.id), new Date(), itemsCleanText, u, eu, g, eg, v, parseFloat(d.totalComm) || 0, parseFloat(d.totalEGP) || 0, parseFloat(d.netProfit) || 0, "مؤكدة" ]; if (targetRow > -1) { sheetSales.getRange(targetRow, 1, 1, 13).setValues([rowValues]); } else { sheetSales.appendRow(rowValues); } } else if (action === "deleteInvoice") { var targetId = parseInt(d.id); var sheetSales = ss.getSheetByName("جدول_البيع") || ss.getSheetByName("Sales"); if (sheetSales) { var sData = sheetSales.getDataRange().getValues(); for (var i = sData.length - 1; i >= 1; i--) { if (parseInt(sData[i][1]) === targetId) { sheetSales.deleteRow(i + 1); } } } } else if (action === "saveFinance") { var sheetFinance = ss.getSheetByName("جدول_المصروفات_والإيرادات") || ss.insertSheet("جدول_المصروفات_والإيرادات"); if (sheetFinance.getLastRow() === 0) { sheetFinance.appendRow(["م", "التاريخ", "النوع", "البيان / الفئة", "المبلغ (EGP)", "ملاحظات"]); } var fRow = [ sheetFinance.getLastRow(), cleanDateStr(d.date), d.type, d.category, parseFloat(d.amount) || 0, d.notes || "" ]; sheetFinance.appendRow(fRow); } else if (action === "deleteFinance") { var sheetFinance = ss.getSheetByName("جدول_المصروفات_والإيرادات"); if (sheetFinance) { var fData = sheetFinance.getDataRange().getValues(); var targetFId = parseInt(d.id); for (var x = fData.length - 1; x >= 1; x--) { if (parseInt(fData[x][0]) === targetFId) { sheetFinance.deleteRow(x + 1); } } } } return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.message })) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } }
