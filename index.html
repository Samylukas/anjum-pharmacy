function doGet(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheetProducts = ss.getSheetByName("جدول_المنتجات") || ss.getSheetByName("Products") || ss.getSheets()[0]; var productsData = sheetProducts ? sheetProducts.getDataRange().getValues() : []; var cloudProducts = []; var productNamesSet = {}; for (var k = 1; k < productsData.length; k++) { var pName = productsData[k][1] ? productsData[k][1].toString().trim() : ""; if (pName && !productNamesSet[pName]) { productNamesSet[pName] = true; cloudProducts.push({ n: pName, p: parseFloat(productsData[k][2]) || 0, q: parseInt(productsData[k][3]) || 0 }); } } var sheetSales = ss.getSheetByName("جدول_البيع") || ss.getSheetByName("Sales"); var salesData = sheetSales ? sheetSales.getDataRange().getValues() : []; var resultInvoices = []; var invoiceIdsSet = {}; var maxId = 1000; for (var i = 1; i < salesData.length; i++) { var invId = parseInt(salesData[i][1]) || 0; if (invId > 0) { if (invId > maxId) maxId = invId; if (!invoiceIdsSet[invId]) { invoiceIdsSet[invId] = true; resultInvoices.push({ id: invId, date: salesData[i][0] || salesData[i][2], itemsText: salesData[i][3] || "", usd: parseFloat(salesData[i][4]) || 0, eur: parseFloat(salesData[i][5]) || 0, gbp: parseFloat(salesData[i][6]) || 0, egp: parseFloat(salesData[i][7]) || 0, visa: parseFloat(salesData[i][8]) || 0, totalComm: parseFloat(salesData[i][9]) || 0, totalEGP: parseFloat(salesData[i][10]) || 0, netProfit: parseFloat(salesData[i][11]) || 0, status: 'Active' }); } } } var output = { status: "success", nextInvoiceId: maxId + 1, products: cloudProducts, invoices: resultInvoices }; var callback = e.parameter ? e.parameter.callback : null; if (callback) { return ContentService.createTextOutput(callback + "(" + JSON.stringify(output) + ")") .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(JSON.stringify(output)).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput("error: " + err.message).setMimeType(ContentService.MimeType.TEXT); } } function doPost(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var req = JSON.parse(e.postData.contents); var action = req.action; var d = req.data; function cleanName(str) { if (!str) return ""; return str.toString().trim().replace(/\s+/g, " "); } function cleanDateStr(dStr) { if (!dStr) return ""; return dStr.toString().replace('T', ' ').replace('.000Z', ''); } if (action === "updateSingleProduct") { var sheetProducts = ss.getSheetByName("جدول_المنتجات") || ss.insertSheet("جدول_المنتجات"); var lastRow = sheetProducts.getLastRow(); if (lastRow === 0) { sheetProducts.appendRow(["م", "اسم المنتج / الصنف", "السعر التجاري (EGP)", "الكمية المتاحة بالمخزن"]); lastRow = 1; } var data = sheetProducts.getRange(1, 1, lastRow, 4).getValues(); var targetName = cleanName(d.n); var found = false; for (var i = 1; i < data.length; i++) { if (cleanName(data[i][1]) === targetName && targetName !== "") { sheetProducts.getRange(i + 1, 4).setValue(d.q); found = true; break; } } if (!found && targetName !== "") { sheetProducts.appendRow([sheetProducts.getLastRow(), d.n, d.p, d.q]); } } else if (action === "saveInvoice") { var sheetSales = ss.getSheetByName("جدول_البيع") || ss.insertSheet("جدول_البيع"); if (sheetSales.getLastRow() === 0) { sheetSales.appendRow(["تاريخ الفاتورة الفعلي", "رقم الفاتورة", "تاريخ الإدخال", "تفاصيل الأصناف المباعة", "دولار ($)", "يورو (€)", "إسترليني (£)", "مصري (EGP)", "فيزا (Visa)", "إجمالي التجاري", "إجمالي البيع (EGP)", "صافي الربح", "الحالة"]); } var salesData = sheetSales.getDataRange().getValues(); var targetRow = -1; for (var k = 1; k < salesData.length; k++) { if (parseInt(salesData[k][1]) === parseInt(d.id)) { targetRow = k + 1; break; } } var itemsCleanText = ""; var u = 0, eu = 0, g = 0, eg = 0, v = 0; if (d.items && Array.isArray(d.items)) { itemsCleanText = d.items.map(function(i) { u += parseFloat(i.usd || 0); eu += parseFloat(i.eur || 0); g += parseFloat(i.gbp || 0); eg += parseFloat(i.egp || 0); v += parseFloat(i.visa || 0); return i.name + " (العدد: " + i.qty + " - بيع: " + (i.sellTotal||0) + " ج.م)"; }).join(" | "); } else if (typeof d.items === 'string') { itemsCleanText = d.items; } var rowValues = [cleanDateStr(d.date), d.id, new Date(), itemsCleanText, u, eu, g, eg, v, d.totalComm || 0, d.totalEGP || 0, d.netProfit || 0, "مؤكدة"]; if (targetRow > -1) { sheetSales.getRange(targetRow, 1, 1, 13).setValues([rowValues]); } else { sheetSales.appendRow(rowValues); } } else if (action === "deleteInvoice") { var targetId = parseInt(d.id); var sheetSales = ss.getSheetByName("جدول_البيع") || ss.getSheetByName("Sales"); if (sheetSales) { var sData = sheetSales.getDataRange().getValues(); for (var i = sData.length - 1; i >= 1; i--) { if (parseInt(sData[i][1]) === targetId) { sheetSales.deleteRow(i + 1); } } } } return ContentService.createTextOutput("Success").setMimeType(ContentService.MimeType.TEXT); } catch (err) { return ContentService.createTextOutput("Error: " + err.message).setMimeType(ContentService.MimeType.TEXT); } }
